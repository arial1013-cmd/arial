  S.group = g;

  });
  render();
}

  const d = S.done.filter(Boolean).length;
  document.getElementById('prog-text').textContent = `${d} / 4 組完成`;
  document.getElementById('prog-fill').style.width = `${d/4*100}%`;
}

function phaseBar(cur, type) {
  const steps = ['📝 選擇題', '✏️ 填空/改寫', '✅ 完成'];
  return steps.map((s,i) => {
    const cls = i < cur ? 'done' : i===cur ? `current${type==='prereq'?' prereq':''}` : 'pending';
    return `<div class="phase-step ${cls}">${i<cur?'✓ ':''}${s}</div>`;
  }).join('');
}

function render() {
  const g = S.group, p = S.phase;
  const grp = GROUPS[g];
  const area = document.getElementById('main-area');
  if(p===0) renderFormula(grp, area);
  else if(p===1) renderMcq(grp, area);
  else if(p===2) renderFill(grp, area);
  else renderDone(g, area);
}

// ── 公式卡 ──
function renderFormula(grp, area) {
  const f = grp.formula, t = grp.type;
  let exHtml = grp.formula.examples.map(e =>
    `<div class="ex-row ${t==='prereq'?'prereq':''}">
      <em>${e.en}</em><span class="ex-note">${e.note}</span>
    </div>`).join('');

  let compareHtml = '';
  if(f.compare) {
    compareHtml = `<div class="compare-grid">${f.compare.map(c=>
      `<div class="cmp ${c.label.includes('🔵')?'pres':'past'}">
        <div class="ct">${c.label}</div>
        <code>${c.lines.map(l=>`<span>${l}</span>`).join('<br>')}</code>
      </div>`).join('')}</div>`;
  }

  let formulaHtml = f.box
    ? `<div class="formula-box">${f.box.replace(/\n/g,'<br>')}</div>`
    : compareHtml;

  area.innerHTML = `
    <div class="phase-bar">${phaseBar(0, t)}</div>
    <div class="formula-card ${t}">
      <div class="formula-title ${t}">${f.title}</div>
      ${formulaHtml}
      <div class="explain">${f.explain.replace(/\n/g,'<br>')}</div>
      ${exHtml}
    </div>
    <button class="next-btn" onclick="S.phase=1;render()">了解了！去做選擇題 →</button>`;
}

// ── 選擇題 ──
function renderMcq(grp, area) {
  const g = grp.id, t = grp.type;
  S.mcqAnswered[g] = 0;
  const html = grp.mcq.map((q,qi) => {
    const optsHtml = q.opts.map((o,oi) =>
      `<button class="mcq-opt" onclick="pickMcq(this,${g},${qi},${oi})">${o}</button>`
    ).join('');
    return `<div class="mcq-block" id="mq${g}_${qi}">
      <div class="mcq-q">
        <div class="q-counter">Q${qi+1} / ${grp.mcq.length}</div>
        <div class="mcq-qtext">${q.q}</div>
        <div class="mcq-options">${optsHtml}</div>
        <div class="feedback" id="mfb${g}_${qi}"></div>
      </div></div>`;
  }).join('');
  area.innerHTML = `
    <div class="phase-bar">${phaseBar(1, t)}</div>
    ${html}
    <button class="next-btn" id="toFill" onclick="S.phase=2;render()" disabled>全部作答後繼續 →</button>`;
}

function pickMcq(btn, g, qi, chosen) {
  const block = document.getElementById(`mq${g}_${qi}`);
  if(block.querySelector('.correct,.wrong')) return;
  const q = GROUPS[g].mcq[qi];
  block.querySelectorAll('.mcq-opt').forEach(o => o.disabled=true);
  const ok = chosen === q.ans;
  btn.classList.add(ok ? 'correct' : 'wrong');
  if(!ok) block.querySelectorAll('.mcq-opt')[q.ans].classList.add('correct');
  const fb = document.getElementById(`mfb${g}_${qi}`);
  fb.className = `feedback ${ok?'ok':'ng'}`;
  fb.innerHTML = ok ? '✓ 正確！' : `✗ 答案是 ${q.opts[q.ans]}。${q.exp ? '<br>'+q.exp : ''}`;
  S.mcqAnswered[g]++;
  if(S.mcqAnswered[g] === GROUPS[g].mcq.length)
    document.getElementById('toFill').disabled = false;
}

// ── 填空/改寫 ──
function renderFill(grp, area) {
  const g = grp.id, t = grp.type;
  S.fillChecked[g] = new Array(grp.fill.length).fill(false);

  const html = grp.fill.map((fq, fi) => {
    const label = fq.type === 'rewrite' ? '✏️ 改寫' : '✏️ 填空';
    return `<div class="fill-block" id="fq${g}_${fi}">
      <div class="fill-label">${label} ${fi+1} / ${grp.fill.length}</div>
      <div class="fill-q">
        <div class="fill-qtext">${fq.q.replace(/\n/g,'<br>')}</div>
        ${fq.hint ? `<div class="fill-hint">${fq.hint}</div>` : ''}
        <input class="fill-input" id="fi${g}_${fi}" type="text" placeholder="${fq.type==='rewrite'?'請輸入完整句子…':'請填入答案…'}" autocomplete="off" autocorrect="off">
        <button class="check-btn" onclick="checkFill(${g},${fi})">確認</button>
        <div class="fill-ans" id="fa${g}_${fi}"></div>
      </div>
    </div>`;
  }).join('');

  area.innerHTML = `
    <div class="phase-bar">${phaseBar(2, t)}</div>
    ${html}
    <button class="next-btn" id="toDone" onclick="finishGroup(${g})" disabled>全部完成 → 過關！</button>`;

  grp.fill.forEach((fq, fi) => {
    document.getElementById(`fi${g}_${fi}`).addEventListener('keydown', e => {
      if(e.key === 'Enter') checkFill(g, fi);
    });
  });
}

function normalise(s) {
  return s.trim().toLowerCase()
    .replace(/['']/g,"'").replace(/[""]/g,'"')
    .replace(/\s+/g,' ');
}

function checkFill(g, fi) {
  const fq = GROUPS[g].fill[fi];
  const input = document.getElementById(`fi${g}_${fi}`);
  const fb = document.getElementById(`fa${g}_${fi}`);
  const val = normalise(input.value);
  if(!val) return;

  const ok = fq.ans.some(a => {
    const na = normalise(a);
    return na === '' ? true : val.includes(na) || na.includes(val);
  });

  input.classList.remove('ok','ng');
  input.classList.add(ok ? 'ok' : 'ng');
  input.disabled = true;
  document.querySelector(`#fq${g}_${fi} .check-btn`).disabled = true;

  fb.className = `fill-ans ${ok?'ok':'ng'}`;
  fb.innerHTML = ok
    ? '✓ 正確！'
    : `✗ 參考答案：<strong>${fq.ans[0]}</strong>`;
  fb.style.display = 'block';

  S.fillChecked[g][fi] = true;
  if(S.fillChecked[g].every(Boolean))
    document.getElementById('toDone').disabled = false;
}

function finishGroup(g) {
  S.done[g] = true;
  if(g < 3) {
    const nt = document.getElementById(`tab-${g+1}`);
    nt.classList.remove('locked');
    const labels = ['先備① 假設語氣','句型① Without/But for','先備② 關係子句','句型② where/when'];
    nt.textContent = labels[g+1];
    if(GROUPS[g+1].type === 'prereq') nt.classList.add('prereq-tab');
  }
  updateProg();
  S.phase = 3;
  render();
}

function renderDone(g, area) {
  const allDone = S.done.every(Boolean);
  const grp = GROUPS[g];
  area.innerHTML = `
    <div class="phase-bar">${phaseBar(3, grp.type)}</div>
    <div class="score-wrap">
      <div class="score-circle">✓</div>
      <div class="score-msg" style="margin-bottom:8px">第 ${g+1} 組「${grp.label}」完成！</div>
      ${allDone ? '<div style="font-size:15px;color:var(--indigo);font-weight:700;margin-top:12px">🎉 四個文法點全部完成！</div>' : ''}
    </div>
    ${g < 3 ? `<button class="unlock-btn" onclick="setGroup(${g+1})">去下一組：${GROUPS[g+1].label} →</button>` : ''}`;
}

updateProg();
render();
